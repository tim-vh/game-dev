## Doolhof structuur opbouwen

Gebruikt de volgende code om een basis structuur te maken om het doolhof te kunnen genereren. Maak in de workspace van Roblox een nieuw script een kopieer de onderstaande code hier naartoe:

```lua
local genStartDelay = 2
local genStructureStepDelay = 0.01
local genMazeStepDelay = 0.5

local mazePosY = 0

local tileSize = 10
local tileHeight = 1
local wallSize = 1
local wallHeight = 10

local mazeMinX = 1
local mazeMaxX = 8
local mazeMinZ = 1
local mazeMaxZ = 8

local tileColor = Color3.fromRGB(163, 162, 165)
local pillarColor = Color3.fromRGB(163, 162, 165)
local wallColor = Color3.fromRGB(163, 162, 165)
local visitedTileColor = Color3.fromRGB(165, 0, 3)

local sections = {}

local direction = {
	North={Name = "North", XDir = 0, ZDir= -1}, 
	South = {Name = "South", XDir = 0, ZDir= 1}, 
	East = { Name = "East", XDir = 1, ZDir= 0}, 
	West = {Name = "West", XDir = -1, ZDir= 0},
	Combine = function(firstDirection, secondDirection)
		return {
			Name = firstDirection.Name .. "-" .. secondDirection.Name,
			XDir = firstDirection.XDir + secondDirection.XDir,
			ZDir = firstDirection.ZDir + secondDirection.ZDir
		}
	end,
}

local posX = 0
local posZ = 0

function pauze(duration)
	if duration ~= 0 then
		task.wait(duration)
	end
end


function createTile(x,z)
	local part = Instance.new("Part", workspace)
	part.Size = Vector3.new(tileSize, tileHeight, tileSize)
	part.Position = Vector3.new(x * (tileSize + wallSize), mazePosY, z * (tileSize + wallSize))
	part.Anchored = true
	part.Material = Enum.Material.SmoothPlastic
	part.Color = tileColor

	return part
end

function createWall(x,z, placeDirection)
	local part = Instance.new("Part", workspace)	
	part.Anchored = true
	part.Material = Enum.Material.SmoothPlastic	
	part.Color = wallColor

	if placeDirection == direction.North or placeDirection == direction.South then
		part.Size = Vector3.new(tileSize, wallHeight + tileHeight, wallSize)
	else
		part.Size = Vector3.new(wallSize, wallHeight + tileHeight, tileSize)
	end

	local xOffset = ((tileSize/ 2) + (wallSize / 2)) * placeDirection.XDir
	local xPos = (x * (tileSize + wallSize)) + xOffset

	local yPos = mazePosY + (wallHeight / 2)

	local zOffset = ((tileSize/ 2) + (wallSize / 2)) * placeDirection.ZDir
	local zPos =  (z * (tileSize + wallSize)) + zOffset

	part.Position = Vector3.new(xPos, yPos , zPos)

	return part
end

function createPillar(x,z, placeDirection)	
	local part = Instance.new("Part", workspace)
	part.Size = Vector3.new(wallSize, wallHeight + tileHeight, wallSize)

	local xOffset = (((tileSize/ 2) + (wallSize / 2)) * placeDirection.XDir )
	local xPos = (x * (tileSize + wallSize)) + xOffset

	local yPos = mazePosY + (wallHeight / 2)

	local zOffset = (((tileSize/ 2) + (wallSize / 2)) * placeDirection.ZDir )
	local zPos = (z * (tileSize + wallSize)) + zOffset

	part.Position = Vector3.new(xPos, yPos, zPos)
	part.Anchored = true
	part.Material = Enum.Material.SmoothPlastic
	part.Color = pillarColor
end

function lowerWall(wall: Part)
	pauze(genMazeStepDelay)

	wall.Size = Vector3.new(wall.Size.X, tileHeight, wall.Size.Z)
	wall.Position = Vector3.new(wall.Position.X, mazePosY, wall.Position.Z)	
end

function createSection(x,z)
	pauze(genStructureStepDelay)

	local section = {}
	section.tile = createTile(x,z)
	section.northWall = nil 
	section.southWall = nil 
	section.eastWall = nil 
	section.westWall = nil 
	section.visited = false

	section.setVisited = function()
		section.visited = true
		section.tile.Color = visitedTileColor
	end

	return section
end


function createMazeStructure()
	pauze(genStartDelay)
	
	for x = mazeMinX, mazeMaxX, 1 do
		sections[x] = {}
		for z = mazeMinZ, mazeMaxZ, 1 do
			sections[x][z] = createSection(x,z)

			sections[x][z].westWall = createWall(x,z, direction.West)
			createPillar(x,z, direction.Combine(direction.North, direction.West))

			if x == mazeMaxX then
				sections[x][z].eastWall = createWall(x,z, direction.East)
				createPillar(x,z, direction.Combine(direction.North, direction.East))
			end

			if x ~= mazeMinX then
				sections[x - 1][z].eastWall = sections[x][z].westWall
			end


			sections[x][z].northWall = createWall(x,z, direction.North)

			if z == mazeMaxZ then
				sections[x][z].southWall = createWall(x,z, direction.South)
				createPillar(x,z, direction.Combine(direction.South, direction.West))
			end

			if z ~= mazeMinZ then
				sections[x][z - 1].southWall = sections[x][z].northWall
			end

			if x == mazeMaxX and z == mazeMaxZ then
				createPillar(x,z, direction.Combine(direction.South, direction.East))
			end

		end
	end
end

function canMoveNorth(x,z)
	if z == mazeMinZ then
		return false
	elseif sections[x][z - 1].visited == true then
		return false
	end

	return true
end

function canMoveSouth(x,z)
	if z == mazeMaxZ then
		return false
	elseif sections[x][z + 1].visited == true then
		return false
	end

	return true
end

function canMoveEast(x,z)
	if x == mazeMaxX then
		return false
	elseif sections[x + 1][z].visited == true then
		return false
	end

	return true
end

function canMoveWest(x,z)
	if x == mazeMinX then
		return false
	elseif sections[x - 1][z].visited == true then
		return false
	end

	return true
end

function goToLocation(x, z)
	posX = x
	posZ = z

	sections[posX][posZ].setVisited()
end

function moveNorth()
	if not canMoveNorth(posX, posZ) then
		warn('cannot move north')
		return
	end
	
	lowerWall(sections[posX][posZ].northWall)

	posZ = posZ - 1
	sections[posX][posZ].setVisited()
end

function moveSouth()
	if not canMoveSouth(posX, posZ) then
		warn('cannot move south')
		return
	end
	
	lowerWall(sections[posX][posZ].southWall)

	posZ = posZ + 1
	sections[posX][posZ].setVisited()
end

function moveEast()
	if not canMoveEast(posX, posZ) then
		warn('cannot move east')
		return
	end
	
	lowerWall(sections[posX][posZ].eastWall)

	posX = posX + 1
	sections[posX][posZ].setVisited()
end

function moveWest()
	if not canMoveWest(posX, posZ) then
		warn('cannot move west')
		return
	end
	
	lowerWall(sections[posX][posZ].westWall)

	posX = posX - 1
	sections[posX][posZ].setVisited()
end
```

Voeg daarna onderaan het script de volgende code toe om de structuur te genereren:

```lua
createMazeStructure()
```

## Pad genereren

Kies eerst op welke plek je wil beginnen met het genereren van het doolhof met deze code:

```lua
goToLocation(1,1)
```

Daarna kun je een pad maken met de functies `moveNorth()`, `moveEast()`, `moveSouth()` en `moveWest()`, bijvoorbeeld op deze manier:

```lua
moveEast()
moveEast()
moveSouth()
moveSouth()
moveWest()
moveNorth()
```

## Controleren of je een bepaalde richting op kan

Je kan controleren of je een bepaalde kant kan opgaan met de functies `canMoveNorth(x,z)`, `canMoveEast(x,z)`, `canMoveSouth(x,z)` en `canMoveWest(x,z). Je kan alleen een kant op als je gaat naar een plek waar je nog niet eerder bent geweest. Als je aan de rand bent van het doolhof kun je ook niet verder die kant op. Met deze functie kun je controleren of je een richting op kan voordat ook daadwerkelijk die richting opgaat. Dit kan bijvoorbeeld op deze manier:

```lua
if canMoveEast() then
	moveEast()
end
```

## Willekeurige richting opgaan

Maak eerst een functie die alle mogelijke richtingen teruggeeft:

```lua
function getPossibleDirections(x,z)
	local directions = {}

	if canMoveNorth(x,z) then
		table.insert(directions, direction.North)
	end
	if canMoveSouth(x,z) then
		table.insert(directions, direction.South)
	end
	if canMoveEast(x,z) then
		table.insert(directions, direction.East)
	end
	if canMoveWest(x,z) then
		table.insert(directions, direction.West)
	end

	return directions
end
```

Daarna kun je met de volgende code een willekeurige kant opgaan:

```lua
local possibleDirections = getPossibleDirections(posX, posZ)

if #possibleDirections > 0 then -- kijk of het aantal richtingen groter is dan 0
	
	local random = math.random(1, #possibleDirections) -- kies een willekeurig getal
	local movingDirection = possibleDirections[random] -- selecteer de richting
	
    -- kijk wat de gekozen richting is en ga die kant op
	if movingDirection == direction.North then
		moveNorth()
	elseif movingDirection == direction.South then
		moveSouth()
	elseif movingDirection == direction.East then
		moveEast()
	elseif movingDirection == direction.West then
		moveWest()
	end
end
```

## Willekeurig pad genereren

Pas de code om een willekeurige kant op te gaan aan door de `if` te vervangen door een `while`. Hiermee blijven we een kant opgaan zolang er een richting beschikbaar is. Onder in de while loop selecteren we weer een nieuwe richting. De code komt er dan zo uit te zien:

```lua
local possibleDirections = getPossibleDirections(posX, posZ)

while #possibleDirections > 0 do -- kijk of het aantal richtingen groter is dan 0

	local random = math.random(1, #possibleDirections) -- kies een willekeurig getal
	local movingDirection = possibleDirections[random] -- selecteer de richting

	-- kijk wat de gekozen richting is en ga die kant op
	if movingDirection == direction.North then
		moveNorth()
	elseif movingDirection == direction.South then
		moveSouth()
	elseif movingDirection == direction.East then
		moveEast()
	elseif movingDirection == direction.West then
		moveWest()
	end
	
	possibleDirections = getPossibleDirections(posX, posZ)
end
```


## Complete doolhof genereren

Om het doolhof compleet te maken moeten we op het moment dat het pad geen richtingen maak op kan een vakje vinden waar nog wel beschikbare richtingen zijn en waar we al een keer zijn geweest. Dit kunnen we doen met de volgende functie:

```lua
function getVisitedStructureWithpossibleDirections(x,z)
	for x = mazeMinX, mazeMaxX, 1 do
		for z = mazeMinZ, mazeMaxZ, 1 do
			if sections[x][z].visited == true then
				local possibleDirections = getpossibleDirections(x,z)
				if #possibleDirections > 0 then
					return Vector3.new(x, 0, z)
				end
			end
		end
	end

	return nil
end
```

Deze functie kunnen we op de volgende manier gebruiken en de bestaande code om een pad te genereren om het volledige doolhof te maken:

```lua
createMazeStructure()

local startLocation = Vector3.new(mazeMinX, 0, mazeMinZ)

while startLocation do 
    goToLocation(startLocation.X, startLocation.Z)
    
    local possibleDirections = getpossibleDirections(posX, posZ)
    while #possibleDirections > 0 do

        local movingDirection = possibleDirections[math.random(1, #possibleDirections)]

        if movingDirection == direction.North then
            moveNorth()
        elseif movingDirection == direction.South then
            moveSouth()
        elseif movingDirection == direction.East then
            moveEast()
        elseif movingDirection == direction.West then
            moveWest()
        end

        possibleDirections = getpossibleDirections(posX, posZ)
    end

    startLocation = getVisitedStructureWithpossibleDirections()
end
```
